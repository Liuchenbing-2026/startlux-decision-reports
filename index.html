是不是普通 Qwen 加提示词？

不是替换成普通 Qwen 权重。本方案仍使用训练好的 StartLux 决策权重和官方提示词；只是复用标准计算接口读取它学到的决策分数。

多个问题分别构造成提示词，再按批次提交；并非把不同问题的概率混在一起。当前实测接口设置 max_tokens=1，会采样一个 token，但决策使用该位置的候选分数，不依赖采样出的字母或长文本解析。

logprobs 与校准，到底做了什么

概念直白解释本方案怎么用
Token（词元）模型处理的编号单位；这里 A–Z 均为单个词元。核验 tokenizer 与 letter_token_ids 一致。
Logit（原始分数）词表输出头给每个词元的未归一化分数。由模型计算，不由服务层编造。
Logprob（对数概率）词表概率取自然对数；越大代表模型越偏好。用 logprob_token_ids 指定所有候选字母。
候选归一化让本题合法选项的概率加起来等于 100%。排除词表中无关词元，不能把缺失候选当作概率为零。
校准温度调整概率集中程度的固定正数，不改变同一题的最高分选项。直接读取该权重发布的参数，不在测试集重新拟合。
pᵢ = exp(ℓᵢ / T) / Σⱼ exp(ℓⱼ / T)　（j 仅遍历本题合法选项）

ℓᵢ 是候选对数概率，T 是对应题型的校准温度。因为完整词表 logprobs 与 logits 只差一个公共常数，这个常数会在候选归一化时抵消。同一组输入分数下两种公式等价；不同数值计算路径不保证逐位一致。

"""Text decisions using untouched vLLM, Ascend and official StartLux helpers."""
import argparse
import json
import math
import sys
import threading
import time
import uuid
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path


def validate_body(body):
    if not isinstance(body, dict):
        raise ValueError('body must be an object')
    if body.get('images'):
        raise ValueError('this service supports text only')
    questions = body.get('questions')
    if not isinstance(questions, dict) or not 1 <= len(questions) <= 16:
        raise ValueError('questions must contain 1..16 named questions')
    for key, question in questions.items():
        if not isinstance(key, str) or not key or not isinstance(question, dict):
            raise ValueError('each question must be a named object')
        kind = question.get('type', 'choice')
        if kind not in ('choice', 'noul', 'score'):
            raise ValueError('supported types: choice, noul, score')
        if not isinstance(question.get('instructions'), str) or not question['instructions'].strip():
            raise ValueError('instructions must be a non-empty string')
        criteria = question.get('criteria')
        if kind == 'choice' and (not isinstance(criteria, dict) or not 2 <= len(criteria) <= 26):
            raise ValueError('choice criteria must contain 2..26 named options')
        if kind == 'score' and (not isinstance(criteria, (dict, list)) or not 2 <= len(criteria) <= 10):
            raise ValueError('score criteria must contain 2..10 levels')
    return questions


class DecisionEngine:
    def __init__(self, model, tp, max_length):
        import torch
        from vllm import LLM, SamplingParams

        # Import the fixed checkpoint's original rendering and answer helpers.
        sys.path.insert(0, str(Path(model).resolve()))
        from startlux_decision import jevfmt
        from startlux_decision.model import StartLuxDecision

        self.torch, self.J, self.answer = torch, jevfmt, StartLuxDecision._answer
        self.name, self.max_prompt_length = Path(model).name, max_length - 1
        cfg = json.loads(Path(model, 'decision_config.json').read_text())
        self.temperatures = cfg['temperature_by_type']
        if any(not math.isfinite(float(t)) or float(t) <= 0 for t in self.temperatures.values()):
            raise ValueError('calibration temperatures must be finite and positive')
        self.llm = LLM(model=model, tensor_parallel_size=tp, dtype='bfloat16',
                       enforce_eager=True, max_model_len=max_length, max_num_seqs=8,
                       max_num_batched_tokens=max_length, gpu_memory_utilization=0.85,
                       enable_prefix_caching=False, enable_chunked_prefill=False,
                       limit_mm_per_prompt={'image': 0, 'video': 0}, seed=0,
                       logprobs_mode='raw_logprobs', generation_config='vllm')
        self.tokenizer = self.llm.get_tokenizer()
        self.letters = self.J.check_tokenizer(self.tokenizer)
        if self.letters != cfg['letter_token_ids']:
            raise ValueError('letter token IDs do not match the checkpoint')
        self.params = SamplingParams(max_tokens=1, temperature=0,
                                     logprob_token_ids=self.letters, detokenize=False)

    def decide(self, body):
        questions = validate_body(body)
        rows = [self.J.from_systemone(body.get('state'), q, key) for key, q in questions.items()]
        prompts = []
        for row in rows:
            ids, _ = self.J.render_ids(row, self.tokenizer, max_length=self.max_prompt_length)
            prompts.append({'prompt_token_ids': ids})
        outputs = self.llm.generate(prompts, self.params, use_tqdm=False)
        if len(outputs) != len(rows):
            raise RuntimeError('incomplete engine response')
        answers, input_tokens, output_tokens = {}, 0, 0
        for (key, question), row, prompt, output in zip(questions.items(), rows, prompts, outputs):
            sample = output.outputs[0]
            logprobs = sample.logprobs[0]
            legal = self.letters[:len(row['options'])]
            if any(token not in logprobs for token in legal):
                raise RuntimeError('engine omitted candidate scores')
            values = [logprobs[token].logprob for token in legal]
            if not all(math.isfinite(value) for value in values):
                raise RuntimeError('non-finite candidate score')
            scores = self.torch.tensor(values, dtype=self.torch.float32)
            temperature = float(self.temperatures[row['type']])
            probabilities = self.torch.softmax(scores / temperature, dim=-1).tolist()
            answers[key] = self.answer(row, probabilities, question)
            input_tokens += len(prompt['prompt_token_ids'])
            output_tokens += len(sample.token_ids)
        return {'answers': answers, 'usage': {'input_tokens': input_tokens, 'output_tokens': output_tokens},
                'model': self.name}


def make_handler(engine):
    lock = threading.Lock()

    class Handler(BaseHTTPRequestHandler):
        def send_json(self, status, body):
            data = json.dumps(body, ensure_ascii=False, allow_nan=False).encode()
            self.send_response(status)
            self.send_header('Content-Type', 'application/json; charset=utf-8')
            self.send_header('Content-Length', str(len(data)))
            self.send_header('x-typesafe-request-id', uuid.uuid4().hex)
            self.end_headers()
            self.wfile.write(data)

        def error(self, status, message):
            self.send_json(status, {'error': message, 'detail': [
                {'loc': ['body'], 'msg': message, 'type': 'value_error'}]})

        def do_GET(self):
            if self.path.rstrip('/') == '/health':
                return self.send_json(200, {'status': 'ok', 'model': engine.name,
                                           'backend': 'vllm-ascend-stock', 'images': False,
                                           'max_prompt_length': engine.max_prompt_length})
            return self.error(404, 'not found')

        def do_POST(self):
            if self.path.rstrip('/') != '/v1/systemone':
                return self.error(404, 'not found')
            try:
                size = int(self.headers.get('Content-Length', '0'))
                if not 0 < size <= 1048576:
                    return self.error(400, 'body must be 1..1048576 bytes')
                body = json.loads(self.rfile.read(size))
            except (ValueError, UnicodeError):
                return self.error(400, 'invalid JSON body')
            start = time.perf_counter()
            try:
                validate_body(body)
                with lock:
                    result = engine.decide(body)
            except (ValueError, TypeError, KeyError) as exc:
                return self.error(422, str(exc))
            except Exception:
                return self.error(500, 'inference failed; inspect the service environment')
            result['latency_ms'] = round((time.perf_counter() - start) * 1000, 2)
            self.send_json(200, result)

        def log_message(self, *_args):
            pass

    return Handler


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--model', required=True)
    parser.add_argument('--host', default='127.0.0.1')
    parser.add_argument('--port', type=int, default=18135)
    parser.add_argument('--tp', type=int, default=2)
    parser.add_argument('--max-model-len', type=int, default=4096)
    args = parser.parse_args()
    engine = DecisionEngine(args.model, args.tp, args.max_model_len)
    server = ThreadingHTTPServer((args.host, args.port), make_handler(engine))
    print(json.dumps({'ready': True, 'model': engine.name, 'port': args.port}), flush=True)
    try:
        server.serve_forever()
    finally:
        server.server_close()


if __name__ == '__main__':
    main()

Perplexity 为什么不直接套这套服务

StartLux 保留标准词表头；Perplexity 原始配置为 Qwen3_5Model，另有形状为 [255, 5120] 的决策读出层，且全注意力要求双向语义。原版加载实测报 language_model 参数映射错误。它需要处理权重映射、决策头与注意力语义；外部插件等不直接改仓库源码的方案可另行验证，但不是此 serve.py 已覆盖的能力。
